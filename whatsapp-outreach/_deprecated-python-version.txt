  - A Chromium-based browser (Chrome / Edge / Brave)
  - WhatsApp Web already logged in (scan QR once)

Usage:
  python send_whatsapp.py              # interactive mode (press Enter to send next)
  python send_whatsapp.py --delay 8    # wait 8s between contacts (lets you click Send)

To customize:
  - Edit CONTACTS below (name, phone in E.164 international format, message)
  - Or pass --csv contacts.csv (columns: name,phone,message)

def load_contacts(csv_path: Path | None) -> list[dict]:
    if csv_path is None:
        return DEFAULT_CONTACTS
    contacts = []
    with csv_path.open(encoding="utf-8") as f:
        for row in csv.DictReader(f):
            row["phone"] = row["phone"].strip()
            row["message"] = row["message"].strip()
            row["name"] = row.get("name", row["phone"]).strip()
            contacts.append(row)
    return contacts


def open_chat(phone: str, message: str) -> None:
    """Open WhatsApp Web chat with the given number and pre-filled message."""
    # Strip non-digits, ensure E.164 starts with +
    digits = "".join(ch for ch in phone if ch.isdigit() or ch == "+")
    if not digits.startswith("+"):
        digits = "+" + digits
    url = f"https://web.whatsapp.com/send?phone={digits}&text={urllib.parse.quote(message)}"
    webbrowser.open(url)


def main() -> int:
    parser = argparse.ArgumentParser(description="Open WhatsApp Web chats in sequence")
    parser.add_argument("--csv", type=Path, help="Optional CSV with columns: name,phone,message")
    parser.add_argument("--delay", type=float, default=4.0, help="Seconds to wait between contacts")
    parser.add_argument("--start", type=int, default=0, help="Start index in the list")
    parser.add_argument("--auto", action="store_true", help="No prompt between contacts (use --delay)")
    args = parser.parse_args()

    contacts = load_contacts(args.csv)
    contacts = contacts[args.start:]

    print(f"\nLoaded {len(contacts)} contact(s).")
    print("Make sure WhatsApp Web is already logged in.")
    print("For each contact, a browser tab will open with the chat pre-filled.")
    print("Click Send in WhatsApp, then come back here and press Enter for the next.\n")

    for i, c in enumerate(contacts, start=args.start):
        print(f"\n[{i + 1}/{len(contacts) + args.start}] {c['name']}")
        print(f"   Phone:   {c['phone']}")
        print(f"   Message: {c['message'][:80]}{'...' if len(c['message']) > 80 else ''}")
        open_chat(c["phone"], c["message"])
        if args.auto:
            time.sleep(args.delay)
        else:
            try:
                input("   Press Enter for next contact (Ctrl+C to quit)...")
            except KeyboardInterrupt:
                print("\nStopped.")
                return 0
    print("\nAll contacts opened. Now click Send in each WhatsApp tab.")
    return 0


if __name__ == "__main__":
    sys.exit(main())
